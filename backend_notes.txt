BACKEND API RELEASE: Here follows the first set of functional API functions implemented in the main branch of AACMS

=======================AACMS API SUITE===========================

=======================FORMAT=======================

EXAMPLE CALL: http://localhost:5000/api/<root>/<call>/:<param>

/api/<root> {
	/<call>/:<param> {
		<param>	 :	call parameter
		type	:	call type
		auth	:	true/false
		desc	:	<description>
		request body {
			"request_1" : "<value 1>",
			"request_2" : "<value 2>"
		}
		response body {
			"response_1" : "<result 1>",
			"response_2" : "<result_2>"
		}
	}
}

=======================USER=======================

/api {
	/users {
		/create {
			type: POST
			auth: false
			desc: Creates a new user account of any supported role

			request body {
				"title" : "Mx/Mr/Ms/etc. - optional, defaults to Mx",
				"first_name" : "user first name",
				"last_name" : "user last name",
				"email" : "user email",
				"password" : "minimum 8 characters",
				"role_id" : 1 - student, 2 - lecturer, 3 - admin

				student role fields {
					"student_number" :
					"study_level" : "Undergraduate/Postgraduate",
					"contact_details" :
					"bank_name" :
					"account_number" :
					"branch_code" :
				}

				lecturer role fields {
					"department" :
				}

				admin role fields {
					"mfa_enabled" : true/false,
					"budget_allocation_rights" : true/false
				}
			}

			response body {
				"message" : "User created successfully.",
				"user_id" : user id,
				"email" : "account email",
				"role_id" : account role
			}
		}

		/login {
			type: POST
			auth: false
			desc: Logs into user account based on email

			request body {
				"email" : "account email",
				"password" : "user password"
			}

			response body {
				"id" : user id,
				"email" : "account email",
				"role_id" : account role

				cookie params {
					"user_id" : user id,
					"role_id" : account role
				}
			}
		}

		/logout {
			type: POST
			auth: false
			desc: Logs user out

			response body {
				"message" : "Process status"
			}
		}

		/fetch/:id {
			param: user id
			type: GET
			auth: true
			desc: Fetches user information based on user ID

			response body {
				"user_id" :
				"title" :
				"first_name" :
				"last_name" :
				"full_name" :
				"email" :
				"role_name" :
				"created_at" :
			}
		}
	}
}
