BACKEND API RELEASE: Here follows the first set of functional API functions implemented in the main branch of AACMS

=======================AACMS API SUITE===========================

=======================FORMAT=======================

EXAMPLE CALL: http://localhost:5000/api/<root>/<call>/:<param>

/api/<root> {
    /<call>/:<param> {
        <param>	 :	call parameter
        type	:	call type
        auth	:	true/false
        desc	:	<description>
        request body {
            "request_1" : "<value 1>",
            "request_2" : "<value 2>"
        }
        response body {
            "response_1" : "<result 1>",
            "response_2" : "<result_2>"
        }
    }
}

=======================USER=======================

/api {
    /users {
        /create {
            type: POST
            auth: false
            desc: Creates a new User account of any type
            request body {
                "first_name" : "user_fname",
                "last_name" : "userser_lname",
                "email" : "user_email",
                "password" : "user password",
                "role_id" : 1 - student, 2 - lecturer, 3 - admin,
                "student_number" : "user_student_number",
                "study_level": "Undergrad/postgrad/staff",
                "contact_details": "user_primary_contact",
                "bank_name": "user_bank_name",
                "account_number": "user_bank_acc",
                "branch_code": "user_branch_code"
            }
            response body {
                "message" : "status message",
                "user_id" : user_id,
                "email" : "account email address",
                "role_id" : account role
            }
        }
        
        /login {
            type: POST
            auth: false
            desc: Logs into user account based on email
            request body {
                "email" : "account email",
                "password" : "user password"
            }
            response body {
                "id" : "user id",
                "email" : "account email",
                "role_id" : account role
                cookie params {
                    "user_id" : "user id",
                    "role_id" : "account role"
                }
            }
        }
        
        /logout {
            type: POST
            auth: false
            desc: Logs user out
            response body {
                "message" : "Process status"
            }
        }
        
        /fetch/:id {
            param: user id
            type: GET 
            auth: true
            desc: responses user information based in ID
            response body {
                "user_id": 17,
                "first_name": "user fname",
                "last_name": "user lname",
                "full_name": "user fullname",
                "email": "account email",
                "role_name": "user role name",
                "created_at": "user created timestamp"
            }
        }

        /perms {
            type: GET
            auth: true
            desc: Returns current user permissions / role access info
            response body {
                "user_id" : user_id,
                "role_id" : account role,
                "permissions" : [ "allowed action(s)" ]
            }
        }
    }			
}

/api {

            response body {
                "stats": {
                    "total_modules": modules in system,

                    "total_demis": assistants appointed,
                    "total_pending_approvals": applications pending approval,
                    "total_pending_claims": claims pending approval
                },
                "monthlyWork": [
                    {
                        "month": "month work was done{yyyy-mm)",
                        "total_sessions": sessions done
                    }
                ],
                "pendingAppointments": [
                    {
                        "application_id": application id 1,
                        "status": "Pending/Approved/Rejected",
                        "date_submitted": "<yyyy-mm-dd> T<hh:mm:ss:ms>Z",
                        "module": "relevant module code",
                        "lecturer": "lecturer title + name",
                        "student": "student title + name"
                    },
                    {
                    ...
                    }
                ]
            }
        }

        /budgets {
            type: GET
            auth: true, admin
            desc: Returns all budgets
            response body {
                "budgets" : [
                    {
                        "budget_id" : budget id,
                        "module_id" : module id,
                        "title" : "budget title",
                        "amount" : amount,
                        "status" : "status",
                        "created_at" : "timestamp"
                    }
                ]
            }
        }

        /budgets/fetch/:budget_id {
            param: budget_id
            type: GET
            auth: true, admin
            desc: Fetches a single budget record by id
            response body {
                "budget_id" : budget id,
                "module_id" : module id,
                "name" : "budget name",
                "amount" : amount,
                "status" : "status"
            }
        }

        /budgets/create {
            type: POST
            auth: true, admin
            desc: Creates a new budget
            request body {
                "title" : "Budget title",
                "module_id" : module id,
                "amount" : 5000,
                "status" : "draft/approved"
            }
            response body {
                "message" : "Budget creation Successful",
                "budget_id" : created_budget_id
            }
        }

        /budgets/edit/:id {
            param: id
            type: PUT
            auth: true, admin
            desc: Updates a budget by id
            request body {
                "title" : "Updated title",
                "amount" : 6000
            }
            response body {
                "message" : "Budget updated successfully",
                "data" : updated_budget_object
            }
        }

        /claims {
            type: GET
            auth: true, admin
            desc: Fetches all claim records for admin review
            response body {
                "claims" : [
                    {
                        "claim_id" : claim id,
                        "status" : "Pending/Approved/Rejected",
                        "amount" : claim amount
                    }
                ]
            }
        }

        /claims/fetch/:claim_id {
            param: claim_id
            type: GET
            auth: true, admin
            desc: Fetches a single claim by id
            response body {
                "claim_id" : claim id,
                "status" : "status",
                "amount" : amount,
                "student_id" : student id
            }
        }
    }	
}
